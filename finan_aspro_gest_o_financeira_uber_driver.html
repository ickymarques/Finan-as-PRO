Script SQL Completo (Supabase)

-- 1. Tabela de Transações
CREATE TABLE IF NOT EXISTS fin_transacoes (
  id TEXT PRIMARY KEY,
  data DATE NOT NULL,
  descricao TEXT NOT NULL,
  categoria TEXT NOT NULL,
  tipo TEXT NOT NULL,
  valor NUMERIC NOT NULL,
  status TEXT DEFAULT 'pago',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Tabela de Metas
CREATE TABLE IF NOT EXISTS fin_metas (
  id TEXT PRIMARY KEY,
  titulo TEXT NOT NULL,
  alvo NUMERIC NOT NULL,
  atual NUMERIC DEFAULT 0,
  prazo DATE,
  icone TEXT DEFAULT '🎯',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Tabela de Recorrências
CREATE TABLE IF NOT EXISTS fin_recorrencias (
  id TEXT PRIMARY KEY,
  descricao TEXT NOT NULL,
  categoria TEXT NOT NULL,
  tipo TEXT NOT NULL,
  valor NUMERIC NOT NULL,
  dia_vencimento INT DEFAULT 1,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Tabela de Turnos Uber
CREATE TABLE IF NOT EXISTS fin_uber_turnos (
  id TEXT PRIMARY KEY,
  data DATE NOT NULL,
  viagens INT DEFAULT 0,
  minutos_online INT DEFAULT 0,
  receita_bruta NUMERIC NOT NULL,
  km NUMERIC NOT NULL,
  consumo_kml NUMERIC NOT NULL,
  preco_litro NUMERIC NOT NULL,
  custo_combustivel NUMERIC NOT NULL,
  reserva_manutencao NUMERIC DEFAULT 0,
  seguro_app NUMERIC DEFAULT 0,
  outros_custos NUMERIC DEFAULT 0,
  lucro_liquido NUMERIC NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Habilitar RLS e Permissões
ALTER TABLE fin_transacoes ENABLE ROW LEVEL SECURITY;
ALTER TABLE fin_metas ENABLE ROW LEVEL SECURITY;
ALTER TABLE fin_recorrencias ENABLE ROW LEVEL SECURITY;
ALTER TABLE fin_uber_turnos ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Acesso total transacoes" ON fin_transacoes FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acesso total metas" ON fin_metas FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acesso total recorrencias" ON fin_recorrencias FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acesso total uber" ON fin_uber_turnos FOR ALL USING (true) WITH CHECK (true);

-- Publicação Realtime
ALTER PUBLICATION supabase_realtime SET TABLE fin_transacoes, fin_metas, fin_recorrencias, fin_uber_turnos;
